// ============================================================
//  zavuch 2 — экспорт ЭЖД-сессии (для dnevnik.mos.ru)
//
//  Скачивает: ejd_session.json
//  Содержит:
//    • cookies — все cookies dnevnik.mos.ru и school.mos.ru
//    • auth_token — из cookies/sessionStorage/localStorage
//    • profile_id — из cookies/localStorage
//    • school_id — если найдём
//
//  Затем этот JSON конвертируется в session.pkl скриптом
//  make_ejd_session.py (см. ниже).
// ============================================================
(async function () {
    "use strict";

    const log = (...a) => console.log("[zavuch2-ejd]", ...a);
    const warn = (...a) => console.warn("[zavuch2-ejd]", ...a);

    // Трекерный мусор — выкидываем
    const SKIP_NAMES = new Set([
        "_ym_d", "_ym_isad", "_ym_uid",
        "tmr_lvid", "tmr_lvidTS",
        "yabm", "GPT_INIT", "sbp_sid",
        "receive-cookie-deprecation",
        "yandex_expboxes", "yashr", "ymex",
        "yp", "yuidss", "skid",
        "pi", "sessar", "Session_id", "sessionid2",
        "yabs-sid",
    ]);

    // ---------- 1. Собираем видимые cookies ----------
    const cookies = {};
    for (const part of document.cookie.split(";")) {
        const [rawName, ...rest] = part.trim().split("=");
        if (!rawName) continue;
        const name = rawName.trim();
        const value = rest.join("=").trim();
        if (!name || SKIP_NAMES.has(name)) continue;
        cookies[name] = value;
    }

    function getCookie(name) {
        const m = document.cookie.match(
            new RegExp("(?:^|; )" +
                name.replace(/([.$?*|{}()[\]\\/+^])/g, "\\$1") + "=([^;]*)")
        );
        return m ? decodeURIComponent(m[1]) : "";
    }

    // ---------- 2. auth_token ----------
    let auth_token = cookies["auth_token"] || getCookie("auth_token") || "";
    if (!auth_token) {
        auth_token =
            sessionStorage.getItem("auth_token") ||
            localStorage.getItem("auth_token") ||
            sessionStorage.getItem("token") ||
            localStorage.getItem("token") ||
            sessionStorage.getItem("access_token") ||
            localStorage.getItem("access_token") ||
            "";
    }
    // Fallback: пробуем вытащить из cookies по имени с точками
    if (!auth_token) {
        for (const n of Object.keys(cookies)) {
            if (n === "auth_token" || n === "token" || n === "access_token") {
                auth_token = cookies[n];
                break;
            }
        }
    }

    // ---------- 3. profile_id ----------
    let profile_id = cookies["profile_id"] || getCookie("profile_id") || "";
    if (!profile_id) {
        profile_id =
            sessionStorage.getItem("profile_id") ||
            localStorage.getItem("profile_id") ||
            "";
    }

    // ---------- 4. school_id (если найдём в storage) ----------
    let school_id = "";
    try {
        school_id =
            sessionStorage.getItem("school_id") ||
            localStorage.getItem("school_id") ||
            "";
    } catch (e) { /* ignore */ }

    // ---------- 5. Итоги ----------
    console.table({
        url: location.href,
        auth_token: auth_token ? `✅ ${auth_token.length} симв.` : "❌ нет",
        profile_id: profile_id || "❌ нет",
        school_id: school_id || "—",
        cookies_visible: Object.keys(cookies).length,
    });

    if (!auth_token) {
        warn(
            "❌ auth_token не найден. Проверьте, что вы залогинены " +
            "на dnevnik.mos.ru, и что токен есть в cookies или localStorage."
        );
    }

    // ---------- 6. Формируем файл ----------
    function nowStr() {
        const d = new Date();
        const p = (n) => String(n).padStart(2, "0");
        return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} `
             + `${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}`;
    }

    const payload = {
        auth_token: auth_token,
        profile_id: profile_id,
        school_id: school_id,
        cookies: cookies,
        saved_at: nowStr(),
        source: "browser_console_ejd",
    };

    // ---------- 7. Скачиваем ----------
    const text = JSON.stringify(payload, null, 2);
    const blob = new Blob([text], { type: "application/json;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = "ejd_session.json";
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    setTimeout(() => URL.revokeObjectURL(url), 3000);

    log("✅ Скачано: ejd_session.json");
    log("");
    log("📌 Дальше:");
    log("   1. Положите файл рядом со скриптом make_ejd_session.py");
    log("   2. Запустите:");
    log("      python make_ejd_session.py ejd_session.json");
    log("   3. Он создаст ~/.zavuch2/session.pkl и auth_data.json.");
})();
